カーネル: atax
サイズ条件: MINI
数値寸法: {"M": 38, "N": 42}

static
void kernel_atax(int m, int n,
		 DATA_TYPE POLYBENCH_2D(A,M,N,m,n),
		 DATA_TYPE POLYBENCH_1D(x,N,n),
		 DATA_TYPE POLYBENCH_1D(y,N,n),
		 DATA_TYPE POLYBENCH_1D(tmp,M,m))
{
  int i, j;

#pragma scop
  /* CPUCOND_LOOP_00 */
for (i = 0; i < _PB_N; i++)
    y[i] = 0;
  /* CPUCOND_LOOP_01 */
for (i = 0; i < _PB_M; i++)
    {
      tmp[i] = SCALAR_VAL(0.0);
      /* CPUCOND_LOOP_02 */
for (j = 0; j < _PB_N; j++)
	tmp[i] = tmp[i] + A[i][j] * x[j];
      /* CPUCOND_LOOP_03 */
for (j = 0; j < _PB_N; j++)
	y[j] = y[j] + A[i][j] * tmp[i];
    }
#pragma endscop

}
