カーネル: bicg
サイズ条件: MEDIUM
数値寸法: {"M": 390, "N": 410}

static
void kernel_bicg(int m, int n,
		 DATA_TYPE POLYBENCH_2D(A,N,M,n,m),
		 DATA_TYPE POLYBENCH_1D(s,M,m),
		 DATA_TYPE POLYBENCH_1D(q,N,n),
		 DATA_TYPE POLYBENCH_1D(p,M,m),
		 DATA_TYPE POLYBENCH_1D(r,N,n))
{
  int i, j;

#pragma scop
  /* CPUCOND_LOOP_00 */
for (i = 0; i < _PB_M; i++)
    s[i] = 0;
  /* CPUCOND_LOOP_01 */
for (i = 0; i < _PB_N; i++)
    {
      q[i] = SCALAR_VAL(0.0);
      /* CPUCOND_LOOP_02 */
for (j = 0; j < _PB_M; j++)
	{
	  s[j] = s[j] + r[i] * A[i][j];
	  q[i] = q[i] + A[i][j] * p[j];
	}
    }
#pragma endscop

}
