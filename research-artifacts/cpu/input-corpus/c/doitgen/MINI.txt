カーネル: doitgen
サイズ条件: MINI
数値寸法: {"NP": 12, "NQ": 8, "NR": 10}

void kernel_doitgen(int nr, int nq, int np,
		    DATA_TYPE POLYBENCH_3D(A,NR,NQ,NP,nr,nq,np),
		    DATA_TYPE POLYBENCH_2D(C4,NP,NP,np,np),
		    DATA_TYPE POLYBENCH_1D(sum,NP,np))
{
  int r, q, p, s;

#pragma scop
  /* CPUCOND_LOOP_00 */
for (r = 0; r < _PB_NR; r++)
    /* CPUCOND_LOOP_01 */
for (q = 0; q < _PB_NQ; q++)  {
      /* CPUCOND_LOOP_02 */
for (p = 0; p < _PB_NP; p++)  {
	sum[p] = SCALAR_VAL(0.0);
	/* CPUCOND_LOOP_03 */
for (s = 0; s < _PB_NP; s++)
	  sum[p] += A[r][q][s] * C4[s][p];
      }
      /* CPUCOND_LOOP_04 */
for (p = 0; p < _PB_NP; p++)
	A[r][q][p] = sum[p];
    }
#pragma endscop

}
