カーネル: gemm
サイズ条件: MINI
数値寸法: {"NI": 20, "NJ": 25, "NK": 30}

static
void kernel_gemm(int ni, int nj, int nk,
		 DATA_TYPE alpha,
		 DATA_TYPE beta,
		 DATA_TYPE POLYBENCH_2D(C,NI,NJ,ni,nj),
		 DATA_TYPE POLYBENCH_2D(A,NI,NK,ni,nk),
		 DATA_TYPE POLYBENCH_2D(B,NK,NJ,nk,nj))
{
  int i, j, k;

//BLAS PARAMS
//TRANSA = 'N'
//TRANSB = 'N'
// => Form C := alpha*A*B + beta*C,
//A is NIxNK
//B is NKxNJ
//C is NIxNJ
#pragma scop
  /* CPUCOND_LOOP_00 */
for (i = 0; i < _PB_NI; i++) {
    /* CPUCOND_LOOP_01 */
for (j = 0; j < _PB_NJ; j++)
	C[i][j] *= beta;
    /* CPUCOND_LOOP_02 */
for (k = 0; k < _PB_NK; k++) {
       /* CPUCOND_LOOP_03 */
for (j = 0; j < _PB_NJ; j++)
	  C[i][j] += alpha * A[i][k] * B[k][j];
    }
  }
#pragma endscop

}
