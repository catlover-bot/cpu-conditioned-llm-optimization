カーネル: jacobi-1d
サイズ条件: MEDIUM
数値寸法: {"N": 400, "TSTEPS": 100}

static
void kernel_jacobi_1d(int tsteps,
			    int n,
			    DATA_TYPE POLYBENCH_1D(A,N,n),
			    DATA_TYPE POLYBENCH_1D(B,N,n))
{
  int t, i;

#pragma scop
  /* CPUCOND_LOOP_00 */
for (t = 0; t < _PB_TSTEPS; t++)
    {
      /* CPUCOND_LOOP_01 */
for (i = 1; i < _PB_N - 1; i++)
	B[i] = 0.33333 * (A[i-1] + A[i] + A[i + 1]);
      /* CPUCOND_LOOP_02 */
for (i = 1; i < _PB_N - 1; i++)
	A[i] = 0.33333 * (B[i-1] + B[i] + B[i + 1]);
    }
#pragma endscop

}
