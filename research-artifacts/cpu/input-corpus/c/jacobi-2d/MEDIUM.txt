カーネル: jacobi-2d
サイズ条件: MEDIUM
数値寸法: {"N": 250, "TSTEPS": 100}

static
void kernel_jacobi_2d(int tsteps,
			    int n,
			    DATA_TYPE POLYBENCH_2D(A,N,N,n,n),
			    DATA_TYPE POLYBENCH_2D(B,N,N,n,n))
{
  int t, i, j;

#pragma scop
  /* CPUCOND_LOOP_00 */
for (t = 0; t < _PB_TSTEPS; t++)
    {
      /* CPUCOND_LOOP_01 */
for (i = 1; i < _PB_N - 1; i++)
	/* CPUCOND_LOOP_02 */
for (j = 1; j < _PB_N - 1; j++)
	  B[i][j] = SCALAR_VAL(0.2) * (A[i][j] + A[i][j-1] + A[i][1+j] + A[1+i][j] + A[i-1][j]);
      /* CPUCOND_LOOP_03 */
for (i = 1; i < _PB_N - 1; i++)
	/* CPUCOND_LOOP_04 */
for (j = 1; j < _PB_N - 1; j++)
	  A[i][j] = SCALAR_VAL(0.2) * (B[i][j] + B[i][j-1] + B[i][1+j] + B[1+i][j] + B[i-1][j]);
    }
#pragma endscop

}
