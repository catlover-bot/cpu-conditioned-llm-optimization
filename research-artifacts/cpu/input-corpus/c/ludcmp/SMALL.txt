カーネル: ludcmp
サイズ条件: SMALL
数値寸法: {"N": 120}

static
void kernel_ludcmp(int n,
		   DATA_TYPE POLYBENCH_2D(A,N,N,n,n),
		   DATA_TYPE POLYBENCH_1D(b,N,n),
		   DATA_TYPE POLYBENCH_1D(x,N,n),
		   DATA_TYPE POLYBENCH_1D(y,N,n))
{
  int i, j, k;

  DATA_TYPE w;

#pragma scop
  /* CPUCOND_LOOP_00 */
for (i = 0; i < _PB_N; i++) {
    /* CPUCOND_LOOP_01 */
for (j = 0; j <i; j++) {
       w = A[i][j];
       /* CPUCOND_LOOP_02 */
for (k = 0; k < j; k++) {
          w -= A[i][k] * A[k][j];
       }
        A[i][j] = w / A[j][j];
    }
   /* CPUCOND_LOOP_03 */
for (j = i; j < _PB_N; j++) {
       w = A[i][j];
       /* CPUCOND_LOOP_04 */
for (k = 0; k < i; k++) {
          w -= A[i][k] * A[k][j];
       }
       A[i][j] = w;
    }
  }

  /* CPUCOND_LOOP_05 */
for (i = 0; i < _PB_N; i++) {
     w = b[i];
     /* CPUCOND_LOOP_06 */
for (j = 0; j < i; j++)
        w -= A[i][j] * y[j];
     y[i] = w;
  }

   /* CPUCOND_LOOP_07 */
for (i = _PB_N-1; i >=0; i--) {
     w = y[i];
     /* CPUCOND_LOOP_08 */
for (j = i+1; j < _PB_N; j++)
        w -= A[i][j] * x[j];
     x[i] = w / A[i][i];
  }
#pragma endscop

}
