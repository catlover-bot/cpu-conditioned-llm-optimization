カーネル: syr2k
サイズ条件: SMALL
数値寸法: {"M": 60, "N": 80}

static
void kernel_syr2k(int n, int m,
		  DATA_TYPE alpha,
		  DATA_TYPE beta,
		  DATA_TYPE POLYBENCH_2D(C,N,N,n,n),
		  DATA_TYPE POLYBENCH_2D(A,N,M,n,m),
		  DATA_TYPE POLYBENCH_2D(B,N,M,n,m))
{
  int i, j, k;

//BLAS PARAMS
//UPLO  = 'L'
//TRANS = 'N'
//A is NxM
//B is NxM
//C is NxN
#pragma scop
  /* CPUCOND_LOOP_00 */
for (i = 0; i < _PB_N; i++) {
    /* CPUCOND_LOOP_01 */
for (j = 0; j <= i; j++)
      C[i][j] *= beta;
    /* CPUCOND_LOOP_02 */
for (k = 0; k < _PB_M; k++)
      /* CPUCOND_LOOP_03 */
for (j = 0; j <= i; j++)
	{
	  C[i][j] += A[j][k]*alpha*B[i][k] + B[j][k]*alpha*A[i][k];
	}
  }
#pragma endscop

}
