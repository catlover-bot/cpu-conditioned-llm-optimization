カーネル: syrk
サイズ条件: SMALL
数値寸法: {"M": 60, "N": 80}

static
void kernel_syrk(int n, int m,
		 DATA_TYPE alpha,
		 DATA_TYPE beta,
		 DATA_TYPE POLYBENCH_2D(C,N,N,n,n),
		 DATA_TYPE POLYBENCH_2D(A,N,M,n,m))
{
  int i, j, k;

//BLAS PARAMS
//TRANS = 'N'
//UPLO  = 'L'
// =>  Form  C := alpha*A*A**T + beta*C.
//A is NxM
//C is NxN
#pragma scop
  /* CPUCOND_LOOP_00 */
for (i = 0; i < _PB_N; i++) {
    /* CPUCOND_LOOP_01 */
for (j = 0; j <= i; j++)
      C[i][j] *= beta;
    /* CPUCOND_LOOP_02 */
for (k = 0; k < _PB_M; k++) {
      /* CPUCOND_LOOP_03 */
for (j = 0; j <= i; j++)
        C[i][j] += alpha * A[i][k] * A[j][k];
    }
  }
#pragma endscop

}
