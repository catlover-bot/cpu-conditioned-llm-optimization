カーネル: trmm
サイズ条件: MEDIUM
数値寸法: {"M": 200, "N": 240}

static
void kernel_trmm(int m, int n,
		 DATA_TYPE alpha,
		 DATA_TYPE POLYBENCH_2D(A,M,M,m,m),
		 DATA_TYPE POLYBENCH_2D(B,M,N,m,n))
{
  int i, j, k;

//BLAS parameters
//SIDE   = 'L'
//UPLO   = 'L'
//TRANSA = 'T'
//DIAG   = 'U'
// => Form  B := alpha*A**T*B.
// A is MxM
// B is MxN
#pragma scop
  /* CPUCOND_LOOP_00 */
for (i = 0; i < _PB_M; i++)
     /* CPUCOND_LOOP_01 */
for (j = 0; j < _PB_N; j++) {
        /* CPUCOND_LOOP_02 */
for (k = i+1; k < _PB_M; k++)
           B[i][j] += A[k][i] * B[k][j];
        B[i][j] = alpha * B[i][j];
     }
#pragma endscop

}
